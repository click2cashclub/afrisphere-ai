function buildLocalTripSummary(
  message: string
): TripSummary {
  const text = message.toLowerCase();

  let travelStyle =
    "Safari & Adventure";

  let bestFor =
    "Wildlife enthusiasts and adventure travellers";

  let accommodation =
    "Safari lodge or tented camp";

  let activities =
    "Wildlife, culture & adventure";

  if (
    text.includes("culture") ||
    text.includes("history") ||
    text.includes("heritage")
  ) {
    travelStyle =
      "Culture & Heritage";

    bestFor =
      "Culture lovers and curious travellers";

    accommodation =
      "Comfortable hotel or cultural lodge";

    activities =
      "Culture, history, food & local experiences";
  }

  if (
    text.includes("romantic") ||
    text.includes("honeymoon") ||
    text.includes("couple")
  ) {
    travelStyle =
      "Romantic Escape";

    bestFor =
      "Couples and honeymoon travellers";

    accommodation =
      "Boutique lodge or luxury safari camp";

    activities =
      "Scenery, dining, relaxation & private experiences";
  }

  if (
    text.includes("family") ||
    text.includes("children") ||
    text.includes("kids")
  ) {
    travelStyle =
      "Family Adventure";

    bestFor =
      "Families and multi-generational travellers";

    accommodation =
      "Family-friendly lodge or hotel";

    activities =
      "Wildlife, sightseeing & family activities";
  }

  if (
    text.includes("budget") ||
    text.includes("cheap") ||
    text.includes("affordable")
  ) {
    return {
      country: "Zimbabwe",
      duration: 5,
      travelStyle,
      transport: "Road transfers",
      budget: "Budget-conscious",
      accommodation:
        "Comfortable guesthouse or budget lodge",
      activities,
      food:
        "Traditional Zimbabwean cuisine",
      transportCost:
        "Varies by route and transport provider",
      dailyEstimate:
        "Depends on accommodation and activities",
      bestFor,
      matchScore: "Excellent Match",

  return {
    country: "Zimbabwe",
    duration: 5,
    travelStyle,
    transport: "Road transfers",
    budget:
      "USD preferred in small crisp bills",
    accommodation,
    activities,
    food:
      "Traditional Zimbabwean cuisine",
    transportCost:
      "Varies by itinerary",
    dailyEstimate:
      "Varies by accommodation and activities",
    bestFor,
    matchScore: "Excellent Match",
  };
}

| POST /api/chat
|--------------------------------------------------------------------------
*/

  try {

    const message =
      typeof body?.message === "string"
        ? body.message.trim()
        : "";

    const history: HistoryMessage[] =
      Array.isArray(body?.history)
        ? body.history
        : [];

    /*
     * Validate message.
     */
    if (!message) {
      return NextResponse.json(
        {
          error: "Message is required.",
          code: "MESSAGE_REQUIRED",
        },
        {
          status: 400,
        }
      );
    }

    if (message.length > 2000) {
      return NextResponse.json(
        {
          error: "Message is too long.",
          code: "MESSAGE_TOO_LONG",
        },
        {
          status: 400,
        }
      );
    }

    /*
     * Keep only recent valid conversation turns.
     */
    const safeHistory = history
      .filter(
        (item) =>
          item &&
          (item.role === "user" ||
            item.role === "zuri") &&
          typeof item.text === "string" &&
          item.text.trim()
      )
      .slice(-20);

    let reply = "";
    let geminiAvailable = false;

    /*
     * ----------------------------------------------------
     * TRY GEMINI
     * ----------------------------------------------------
     */

    if (apiKey) {
      try {
        const priorTurns =
          safeHistory.map((item) => ({
            role:
              item.role === "user"
                ? "user"
                : "model",
            parts: [
              {
                text: item.text.trim(),
              },
            ],
          }));

        const contents = [
          ...priorTurns,
          {
            role: "user" as const,
            parts: [
              {
                text: message,
              },
            ],
          },
        ];

        const ai = new GoogleGenAI({
          apiKey,
        });

        const response =
          await ai.models.generateContent({
            model: "gemini-3.6-flash",
            contents,
            config: {
              systemInstruction:
                ZURI_SYSTEM_INSTRUCTION,
              temperature: 0.7,
            },
          });

        reply =
          response.text?.trim() ?? "";

        if (reply) {
          geminiAvailable = true;
        }
      } catch (error: any) {
        const status =
          typeof error?.status === "number"
            ? error.status
            : typeof error?.statusCode ===
                "number"
              ? error.statusCode
              : 500;

        const errorMessage =
          typeof error?.message === "string"
            ? error.message
            : "";

        const quotaError =
          status === 429 ||
          errorMessage.includes(
            "RESOURCE_EXHAUSTED"
          ) ||
          errorMessage
            .toLowerCase()
            .includes("quota") ||
          errorMessage
            .toLowerCase()
            .includes("rate limit");

        if (quotaError) {
          console.warn(
            "Gemini quota reached. Switching Zuri to local fallback mode."
          );
        } else {
          console.error(
            "Gemini generation failed. Switching Zuri to local fallback mode:",
            error
          );
        }
      }
    } else {
      console.warn(
        "GEMINI_API_KEY is missing. Using local Zuri fallback mode."
      );
    }

    /*
     * ----------------------------------------------------
     * LOCAL ANSWER FALLBACK
     * ----------------------------------------------------
     */

    if (!geminiAvailable) {
      reply = buildLocalFallback(message);
    }

    /*
     * ----------------------------------------------------
     * LOCAL STRUCTURED DATA
     * ----------------------------------------------------
     *
     * This is always generated locally.
     *
     * That means hotels, restaurants, destinations and
     * experiences continue working even if Gemini is down.
     */

    const localStructured =
      buildLocalStructuredData(
        message,
        reply
      );

    const localHotels =
      getLocalHotels(message);

    const localRestaurants =
      getLocalRestaurants(message);

    const localTrip =
      buildLocalTripSummary(message);

    /*
     * ----------------------------------------------------
     * STRUCTURED GEMINI EXTRACTION
     * ----------------------------------------------------
     *
     * IMPORTANT:
     *
     * Do NOT call extractTravelData when Gemini itself
     * already failed. Otherwise we would make a second
     * Gemini request and potentially hit the same quota.
     */

    let structuredData = {
      trip: {} as TripSummary,
      destinations: [] as Destination[],
      hotels: [] as Recommendation[],
      restaurants: [] as Recommendation[],
      experiences: [] as Experience[],
      followUpQuestions: [] as string[],
    };

    if (geminiAvailable) {
      try {
        const extracted =
          await extractTravelData(reply);

        structuredData = {
          trip: extracted.trip ?? {},
          destinations:
            extracted.destinations ?? [],
          hotels:
            extracted.hotels ?? [],
          restaurants:
            extracted.restaurants ?? [],
          experiences:
            extracted.experiences ?? [],
          followUpQuestions:
            extracted.followUpQuestions ?? [],
        };
      } catch (error) {
        console.error(
          "Structured travel extraction failed:",
          error
        );
      }
    }

    /*
     * ----------------------------------------------------
     * MERGE STRUCTURED RESULTS
     * ----------------------------------------------------
     */

    const destinations =
      structuredData.destinations.length > 0
        ? structuredData.destinations
        : localStructured.destinations;

    const hotels =
      structuredData.hotels.length > 0
        ? structuredData.hotels
        : localHotels;

    const restaurants =
      structuredData.restaurants.length > 0
        ? structuredData.restaurants
        : localRestaurants;

    const experiences =
      structuredData.experiences.length > 0
        ? structuredData.experiences
        : localStructured.experiences;

    const trip: TripSummary = {
      ...localTrip,
      ...structuredData.trip,
    };

    /*
     * Make sure important fields never disappear.
     */
    if (!trip.country) {
      trip.country = localTrip.country;
    }

    if (!trip.duration) {
      trip.duration = localTrip.duration;
    }

    if (!trip.travelStyle) {
      trip.travelStyle =
        localTrip.travelStyle;
    }

    if (!trip.transport) {
      trip.transport =
        localTrip.transport;
    }

    if (!trip.budget) {
      trip.budget = localTrip.budget;
    }

    if (!trip.accommodation) {
      trip.accommodation =
        localTrip.accommodation;
    }

    if (!trip.activities) {
      trip.activities =
        localTrip.activities;
    }

    if (!trip.food) {
      trip.food = localTrip.food;
    }

    if (!trip.transportCost) {
      trip.transportCost =
        localTrip.transportCost;
    }

    if (!trip.dailyEstimate) {
      trip.dailyEstimate =
        localTrip.dailyEstimate;
    }

    if (!trip.bestFor) {
      trip.bestFor =
        localTrip.bestFor;
    }

    if (!trip.matchScore) {
      trip.matchScore =
        localTrip.matchScore;
    }

    /*
     * Follow-up questions.
     */
    const followUpQuestions =
      structuredData.followUpQuestions
        .filter(
          (question) =>
            typeof question === "string" &&
            question.trim()
        )
        .slice(0, 3);

    if (followUpQuestions.length === 0) {
      followUpQuestions.push(
        "Would you prefer a budget, comfortable or luxury itinerary?"
      );
    }

    /*
     * ----------------------------------------------------
     * FINAL RESPONSE
     * ----------------------------------------------------
     */

    const responseData: ChatResponse = {
      answer: reply,

      trip,

      destinations,

      hotels,

      restaurants,

      experiences,

      followUpQuestions,
    };

    return NextResponse.json(
      responseData,
      {
        status: 200,
      }
    );
  } catch (error) {
    /*
     * ----------------------------------------------------
     * FINAL SAFETY NET
     * ----------------------------------------------------
     */

    console.error(
      "Zuri API error:",
      error
    );

    const fallbackMessage =
      "Zimbabwe travel";

    const fallbackAnswer =
      buildLocalFallback(
        fallbackMessage
      );

    const fallbackStructured =
      buildLocalStructuredData(
        fallbackMessage,
        fallbackAnswer
      );

    const fallbackHotels =
      getLocalHotels(
        fallbackMessage
      );

    const fallbackRestaurants =
      getLocalRestaurants(
        fallbackMessage
      );

    const fallbackTrip =
      buildLocalTripSummary(
        fallbackMessage
      );

    return NextResponse.json(
      {
        answer: fallbackAnswer,

        trip: fallbackTrip,

        destinations:
          fallbackStructured.destinations,

        hotels: fallbackHotels,

        restaurants:
          fallbackRestaurants,

        experiences:
          fallbackStructured.experiences,

        followUpQuestions: [
          "Would you like me to build a detailed Zimbabwe itinerary?",
        ],
      },
      {
        status: 200,
      }
    );
  }
}
